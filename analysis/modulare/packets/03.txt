  34  function gleichWeit() {
  35    const x = rnd(3, 12);
  36    return num(x, `Der Punkt P liegt auf der Mittelsenkrechten der Strecke AB. Es ist PA = ${x} cm. Wie lang ist PB?`,
  37      `Jeder Punkt der Mittelsenkrechten ist von A und B gleich weit entfernt. Also ist PB = PA = ${x} cm.`, 'Rechne', [x * 2, x / 2, x + 1], ' cm');
  38  }

  40  function mitteStrecke() {
  41    const h = rnd(2, 9);
  42    return num(h, `Die Strecke AB ist ${2 * h} cm lang. Die Mittelsenkrechte schneidet AB im Punkt M. Wie lang ist AM?`,
  43      `M ist der Mittelpunkt der Strecke. AM = ${2 * h} cm : 2 = ${h} cm.`, 'Rechne', [2 * h, h + 2, h * 2 + 1], ' cm');
  44  }

  49      `mehr als ${h} cm`, [`genau ${h} cm`, `weniger als ${h} cm`, `genau ${ab * 2} cm, sonst nichts`],
  50      `Die Kreisbögen um A und B müssen sich in zwei Punkten schneiden. Das klappt nur, wenn der Radius größer ist als die Hälfte der Strecke (${h} cm). Bei genau ${h} cm berühren sich die Bögen nur in M, bei weniger treffen sie sich gar nicht.`, 'Konstruktion');

  71  function dreieckSeiten() {
  72    let a, b, c;
  73    do { a = rnd(3, 9); b = rnd(3, 9); c = rnd(3, 9); } while (a + b <= c || a + c <= b || b + c <= a);
  74    return words(`Ein Dreieck hat die Seiten a = ${a} cm, b = ${b} cm und c = ${c} cm. Wie viele Dreiecke gibt es dazu?`, 'genau eines (bis auf Spiegelung und Drehung)',
  75      ['kein einziges', 'unendlich viele', 'genau zwei verschiedene'],
  76      `Bei drei Seiten (SSS) gibt es höchstens ein Dreieck, und weil jede Seite kürzer ist als die beiden anderen zusammen, gibt es auch eines.`, 'Eindeutigkeit');
  77  }

 103  const ANGLE_KINDS = [
 104    ['Scheitelwinkel', (a) => a, 'Scheitelwinkel liegen sich gegenüber und sind gleich groß.'],
 105    ['Nebenwinkel', (a) => 180 - a, 'Nebenwinkel ergeben zusammen 180°.'],

 110  function winkelPaar() {
 111    const [kind, fn, why] = pick(ANGLE_KINDS), a = rnd(2, 17) * 10, v = fn(a);
 112    const needsParallel = kind === 'Stufenwinkel' || kind === 'Wechselwinkel';
 113    return num(v, `${needsParallel ? 'Zwei parallele Geraden werden von einer dritten geschnitten.' : 'Zwei Geraden schneiden sich.'} Ein Winkel misst ${a}${D}. Wie groß ist der ${kind}?`,
 114      `${why} Also ${fmt(v)}${D}.`, 'Winkel', [180 - a, a, 90 - (a % 90)].filter((x) => x > 0), D);
 115  }

 117  function winkelsumme() {

 121  }
