   1  // Dashboard: liest Profil und Fortschritt aller Apps aus dem Browser-Speicher und zeigt sie als Kacheln.
   2  import { esc, $ } from './core/js/ui.js';
   3  import { mascot } from './core/js/mascot.js';
   4  import { createStore } from './core/js/store.js';
   5  import { GOAL, openProfile, appKey, appDefaults, dailyXP, streakNow } from './core/js/profile.js';
   6  import { APPS, SUBJECT_COLORS } from './registry.js';
   7  
   8  // Der Browser-Speicher wird asynchron geöffnet. Erst wenn alles geladen ist, wird gezeichnet.
   9  async function load() {
  10    const profile = (await openProfile()).state;
  11    const rows = await Promise.all(APPS.map(async (app) => ({
  12      app,
  13      state: app.status === 'live' ? (await createStore(appKey(app.id), appDefaults())).state : null
  14    })));
  15    start(profile, rows);
  16  }
  17  
  18  function start(profile, rows) {
  19    const doneUnits = (row) => Object.values(row.state.stars).filter((n) => n > 0).length;
  20    const colorOf = (app) => SUBJECT_COLORS[app.subject] || '#9B7BFF';
  21    const badgeCount = rows.reduce((sum, r) => sum + (r.state ? Object.keys(r.state.badges).length : 0), 0);
  22  
  23    function renderHero() {
  24      const x = dailyXP(profile), ratio = Math.min(1, x / GOAL), C = 2 * Math.PI * 32;
  25      const says = x >= GOAL ? 'Tagesziel geschafft. Alles ab jetzt ist Bonus.'
  26        : x === 0 ? 'Welche App heute? Fünf Minuten reichen schon.' : `Noch ${GOAL - x} XP bis zum Tagesziel.`;
  27      $('#hero').innerHTML = `
  28        <div class="row1">${mascot('happy')}<div><h1>${profile.name ? 'Hallo, ' + esc(profile.name) : 'Hallo'} !</h1><div class="bubble" style="margin-top:10px">${esc(says)}</div></div></div>
  29        <div class="stats" style="gap:14px 24px;flex-wrap:wrap">
  30          <div class="ring" role="img" aria-label="Tagesziel ${Math.min(x, GOAL)} von ${GOAL} XP"><svg viewBox="0 0 76 76"><circle class="t" cx="38" cy="38" r="32" fill="none" stroke-width="8"/><circle class="p" cx="38" cy="38" r="32" fill="none" stroke-width="8" stroke-linecap="round" stroke-dasharray="${C.toFixed(1)}" stroke-dashoffset="${(C * (1 - ratio)).toFixed(1)}"/></svg><b>${Math.min(x, GOAL)}<br><small style="font-size:11px;font-weight:600">/ ${GOAL}</small></b></div>
  31          <div class="stat"><b>${streakNow(profile)}</b><span>Tage in Folge</span></div>
  32          <div class="stat"><b>${profile.xp}</b><span>XP gesamt</span></div>
  33          <div class="stat"><b>${badgeCount}</b><span>Trophäen</span></div>
  34        </div>`;
  35    }
  36  
  37    function renderContinue() {
  38      const row = rows.find((r) => r.app.id === profile.lastApp && r.app.status === 'live');
  39      const section = $('#cont-sec');
  40      if (!row) { section.hidden = true; return; }
  41      section.hidden = false;
  42      section.innerHTML = `<div class="card cont" style="--c:${colorOf(row.app)}"><span class="dot"></span><div><h3>Weiter mit ${esc(row.app.title)}</h3><p class="muted">${doneUnits(row)} von ${row.app.units} Seiten · ${row.state.xp} XP</p></div><a class="btn sun" href="${esc(row.app.href)}">Weiterlernen</a></div>`;
  43    }
  44  
  45    function tile(row) {
  46      const { app, state } = row, color = colorOf(app);
  47      if (app.status !== 'live') {
  48        return `<div class="card tile soon" style="--c:${color}"><span class="tag">${esc(app.subject)}</span><h3>${esc(app.title)}</h3><p>${esc(app.desc)}</p><span class="pill">Demnächst</span></div>`;
  49      }
  50      const done = doneUnits(row), pct = Math.round((done / app.units) * 100);
  51      const info = state.xp || done ? `${done} von ${app.units} Seiten · ${state.xp} XP` : 'Noch nicht gestartet';
  52      return `<a class="card tile" style="--c:${color}" href="${esc(app.href)}"><span class="tag">${esc(app.subject)}</span><h3>${esc(app.title)}</h3><p>${esc(app.desc)}</p><div class="lab">${info}</div><div class="bar"><i style="width:${pct}%"></i></div></a>`;
  53    }
  54  
  55    const subjects = ['Alle', ...new Set(APPS.map((a) => a.subject))];
  56    let current = 'Alle';
  57  
  58    function renderGrid() {
  59      $('#chips').innerHTML = subjects.map((s) => `<button class="chip" type="button" data-s="${esc(s)}" aria-pressed="${s === current}">${esc(s)}</button>`).join('');
  60      const list = rows.filter((r) => current === 'Alle' || r.app.subject === current);
  61      $('#empty').style.display = list.length ? 'none' : 'block';
  62      $('#grid').innerHTML = list.map(tile).join('');
  63    }
  64  
  65    $('#chips').addEventListener('click', (e) => {
  66      const button = e.target.closest('[data-s]');
  67      if (!button) return;
  68      current = button.dataset.s;
  69      renderGrid();
  70    });
  71  
  72    renderHero();
  73    renderContinue();
  74    renderGrid();
  75  }
  76  
  77  load();
