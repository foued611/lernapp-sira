  12  test('Deutsch: jede Aufgabe ist in sich stimmig', () => {
  13    for (const [unit, list] of items) {
  14      assert.ok(list.length >= 10, `${unit}: zu wenige Aufgaben`);
  15      assert.equal(new Set(list.map((i) => `${i.q}|${i.s || ''}|${i.a}`)).size, list.length, `${unit}: doppelte Aufgabe`);
  16      for (const item of list) {
  17        const id = `${unit}: ${item.q} ${item.s || ''}`;
  18        assert.ok(item.why && item.why.length > 10, `${id}: Erklärung fehlt`);
  19        assert.ok(!item.wrong.includes(item.a), `${id}: richtige Antwort steht bei den falschen`);
  20        assert.equal(new Set(item.wrong).size, item.wrong.length, `${id}: falsche Antworten doppelt`);
  21        if (item.kind === 'gap') {
  22          assert.equal(item.s.split('___').length, 2, `${id}: genau eine Lücke`);
  23          assert.ok(item.wrong.length >= 1);
  24        } else {
  25          assert.ok(item.wrong.length >= 3, `${id}: mindestens drei falsche Antworten`);
  26        }
  27      }
  28    }
  29  });

  31  test('Deutsch: Lücken-Aufgaben lassen sich auswählen und eintippen', () => {
  32    for (const [, list] of items) for (const item of list.filter((i) => i.kind === 'gap')) {
  33      const c = choice(item);
  34      assertValidQuestion(c);
  35      assert.ok(c.options.some((o) => o.id === item.a && o.fill === item.a));
  36      const t = typed(item);
  37      assertValidQuestion(t);
  38      assert.equal(t.check(item.a), 'ok');
  39      // Wechselnde Groß-/Kleinschreibung statt toUpperCase(): Das macht aus „ß“ ein „SS“ und würde die Aufgabe verfälschen.
  40      const mixedCase = [...item.a].map((c, i) => (c === 'ß' ? c : i % 2 ? c.toLowerCase() : c.toUpperCase())).join('');
  41      assert.equal(t.check(` ${mixedCase} `), 'ok', 'Groß/Klein und Leerzeichen werden toleriert');
  42      assert.equal(t.check(item.wrong[0]), 'bad');
