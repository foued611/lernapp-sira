   1  // Zustand einer Einheit (Profil, App): im Speicher schnell lesbar, in der Browser-Datenbank dauerhaft.
   2  // Flexibles Schema: gespeicherte Felder bleiben erhalten, neue Standardfelder werden beim Laden ergänzt.
   3  // Eine Migration ist nur nötig, wenn sich die Bedeutung vorhandener Felder ändert.
   4  import { getDB } from './db.js';

   6  const isPlain = (v) => v && typeof v === 'object' && !Array.isArray(v);
   7  const clone = (v) => JSON.parse(JSON.stringify(v));
   8  
   9  /** Führt gespeicherte Daten über die Standardwerte, auch in verschachtelten Objekten. */
  10  export function deepMerge(base, data) {
  11    const out = { ...base };
  12    for (const [key, value] of Object.entries(data)) {
  13      out[key] = isPlain(value) && isPlain(base[key]) ? deepMerge(base[key], value) : value;
  14    }
  15    return out;
  16  }

  18  /**
  19   * Lädt (oder legt an) ein Dokument. Danach ist state synchron nutzbar, save() schreibt im Hintergrund.
  20   * @param {string} key        eindeutiger Schlüssel, z. B. "lernwerkstatt:app:fr-verben"
  21   * @param {object} defaults   Startzustand
  22   * @param {{version?:number, migrate?:(old:object, fromVersion:number)=>object}} [options]
  23   */
  24  export async function createStore(key, defaults, { version = 1, migrate } = {}) {
  25    const db = await getDB();
  26    let stored;
  27    try { stored = await db.get(key); } catch (error) { stored = undefined; }
  28    if (stored === undefined) {
  29      stored = await db.legacy(key);
  30      if (stored !== undefined && !db.sync) { try { await db.put(key, stored); } catch (error) { /* wird beim nächsten save() erneut versucht */ } }
  31    }
  32  
  33    const fresh = () => clone(defaults);
  34    let state = fresh();
  35    if (isPlain(stored)) {
  36      try {
  37        let data = clone(stored);
  38        const from = data.__v || 1;
  39        if (from !== version && migrate) data = migrate(data, from);
  40        delete data.__v;
  41        state = deepMerge(fresh(), data);
  42      } catch (error) {
  43        state = fresh();
  44      }
  45    }
  46  
  47    const store = {
  48      state,
  49      backend: db.name,
  50      save() {
  51        db.put(key, { ...store.state, __v: version }).catch(() => { /* Speicher voll oder gesperrt: die App läuft weiter */ });
  52      },
  53      reset() {
  54        store.state = fresh();
  55        store.save();
  56      }
  57    };
  58    return store;
  59  }
