  36  export function placeValueQ() {
  37    const kind = rnd(0, 2);
  38    const n = rnd(100, 9999);
  39    const d = placeDigits(n);
  40    const names = ['Tausender', 'Hunderter', 'Zehner', 'Einer'];
  41    if (kind === 0) {
  42      const i = rnd(0, 3);
  43      const digit = d[i];
  44      const value = String(Number(digit) * [1000, 100, 10, 1][i]);
  45      return choice({
  46        label: 'Stellenwert erkennen',
  47        prompt: `Welchen Wert hat die <b>${digit}</b> in ${mono(String(n))}?`,
  48        promptStyle: 'sentence',
  49        answer: value,
  50        wrong: [Number(digit) * 10 * [1000, 100, 10, 1][i], Number(digit), Number(digit) * [1000, 100, 10, 1][Math.min(3, i + 1)], Number(digit) * [1000, 100, 10, 1][Math.max(0, i - 1)]],
  51        why: `Die ${digit} steht an der ${names[i]}-Stelle (${names[i]} = ${[1000, 100, 10, 1][i]}). Also ${digit} × ${[1000, 100, 10, 1][i]} = ${value}.`
  52      });
  53    }
  54    if (kind === 1) {
  55      return choice({
  56        label: 'Zahl zerlegen',
  57        prompt: `Zerlege ${mono(String(n))}:`,
  58        promptStyle: 'sentence',
  59        answer: placeSum(n),
  60        wrong: [placeSum(Number(d.slice().reverse().join(''))), placeSum(n + 10), placeSum(n - 100), placeSum(Number(d[1] + d[0] + d[3] + d[2]))],
  61        why: `Jede Ziffer wird mit ihrem Stellenwert multipliziert: ${placeSum(n)}.`
  62      });
  63    }
  64    return typed({
  65      label: 'Zahl zusammensetzen',
  66      prompt: `${placeSum(n).replace(/×/g, ' × ')} = ?`,
  67      answer: String(n),
  68      type: 'dec',
  69      why: `Rechne jeden Teil aus und addiere: ${placeSum(n)} = ${n}.`,
  70      placeholder: 'Zahl eintippen'
  71    });
  72  }

  77    const [lo, hi] = RANGE[level];

  79    const bin = toBin(n);

  85        answer, wrong: [parseInt(reversed(bin), 2), parseInt(flipBits(bin), 2), n + 1, n - 1, n + 2],
