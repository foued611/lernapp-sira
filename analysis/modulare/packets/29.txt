   9    `<span class="stars" aria-label="${n} von 3 Sternen">${[1, 2, 3].map((i) => `<span class="${i <= n ? 'on' : ''}">${icon('star')}</span>`).join('')}</span>`;

  11  export function minouSays(profile) {
  12    const x = dailyXP(profile);
  13    if (x >= GOAL) return 'Tagesziel geschafft. Alles ab jetzt ist Bonus.';
  14    if (x === 0) return pick(['Nur 2 Minuten, dann bist du schon im Spiel.', 'Eine kurze Runde reicht für heute.', 'Klein anfangen zählt. Fünf Aufgaben, mehr nicht.']);
  15    return `Noch ${GOAL - x} XP bis zum Tagesziel. Das ist ungefähr eine Runde.`;
  16  }
