  15  export function placeSum(n) {

  17    const parts = [[1000, d[0]], [100, d[1]], [10, d[2]], [1, d[3]]].filter((p) => p[1] !== '0').map((p) => `${p[1]}×${p[0]}`);

  21  /** Stellenwerte, über denen in der Binärzahl eine 1 steht. */
  22  export function onesWeights(bin) {
  23    const len = bin.length;
  24    return bin.split('').map((c, i) => (c === '1' ? 2 ** (len - 1 - i) : 0)).filter(Boolean);
  25  }

  27  /** Schritte beim Umrechnen von Normal nach Binär: "Passt der Stellenwert noch hinein?" */
  28  export function decToBinSteps(n) {
  29    const steps = [];
  30    let rest = n;
  31    let started = false;
  32    for (const w of WEIGHTS) {
  33      if (!started && w > n && w !== 1) continue;
  34      started = true;
  35      const fits = w <= rest;
  36      steps.push({ w, rest, fits });
  37      if (fits) rest -= w;
  38    }
  39    return steps;
  40  }

  65  /** Eingabe vereinheitlichen: ohne Leerzeichen, ohne 0x, ohne führende Nullen, Hex in Großbuchstaben. */
  66  export function normInput(raw, type) {
  67    const s = String(raw).trim().toUpperCase().replace(/[\s_]+/g, '').replace(/^0X/, '');
  68    if (type === 'dec') return /^\d+$/.test(s) ? String(parseInt(s, 10)) : null;
  69    if (type === 'bin') return /^[01]+$/.test(s) ? s.replace(/^0+(?=.)/, '') : null;
  70    return /^[0-9A-F]+$/.test(s) ? s.replace(/^0+(?=.)/, '') : null;
  71  }

  76  /** Falsche Antworten für Auswahlfragen, ohne Dopplungen und ohne die richtige Antwort. */
  77  export function distractors(answer, candidates, count = 3) {
  78    const seen = new Set([String(answer)]);
  79    const out = [];
  80    for (const c of candidates) {
  81      const v = String(c);
  82      if (!seen.has(v) && v !== 'NaN' && !v.startsWith('-')) { seen.add(v); out.push(v); }
  83      if (out.length === count) return out;
  84    }
  85    // Auffüllen mit Nachbarzahlen, falls zu wenige Kandidaten da waren
  86    const num = parseInt(answer, 10);
  87    for (let d = 1; out.length < count && !Number.isNaN(num); d++) {
  88      for (const v of [String(num + d), String(num - d)]) {
  89        if (!seen.has(v) && !v.startsWith('-') && out.length < count) { seen.add(v); out.push(v); }
  90      }
  91    }
  92    return out;
  93  }
