   1  // Universeller Quiz-Motor: Runde mit Herzen, Combo, XP, Spickzettel und Feedback sowie der Blitz-Modus.
   2  // Er kennt keinen Lernstoff. Jede App liefert Fragen im folgenden Format.
   3  //
   4  // Frage = {
   5  //   kind:   'choice' | 'type' | 'match' (oder ein eigener Typ aus core/js/qtypes.js)
   6  //   label:  string               kleine Überschrift, z. B. "Lücke füllen"
   7  //   chip?:  html                 kleines Info-Feld über der Frage
   8  //   prompt?: html                Haupttext, darf {{blank}} als Lücke enthalten
   9  //   promptStyle?: 'sentence' | 'big'
  10  //   sub?:   string               graue Zeile (z. B. Übersetzung), erscheint bei Erfolg als "= …"
  11  //   answer: string               choice: Option-id, type: erwarteter Text
  12  //   options?: [{ id, html, fill? }]   choice; fill = Text für die Lücke
  13  //   optionsLayout?: 'list'       Optionen untereinander (für ganze Sätze)
  14  //   placeholder?: string, accents?: boolean (Zeichenleiste aus ctx.inputChars, Standard an), check?: (input) => 'ok'|'almost'|'bad'   type
  15  //   left?: [html], right?: [{ value, html }], leftValues?: [value]                    match
  16  //   reveal?: html                Korrektur bei Fehler (Stiftschrift)
  17  //   why?: string                 kurze Erklärung bei Fehler
  18  //   help?: html                  Inhalt des Spickzettels

  39  function feedbackHTML(q, run, ctx) {
  40    const f = run.fb;
  41    const next = '<button class="btn" data-act="next" id="nx">Weiter</button>';
  42    if (f.kind === 'ok') {
  43      const praise = f.retry ? 'Diesmal sitzt es!' : f.mult > 1 ? `Serie ×${f.mult}!` : pick(ctx.words.praise);
  44      return `<div class="fb ok" role="status">${mascot(f.mult > 1 ? 'cheer' : 'happy')}<div class="msg"><b>${praise}</b> <span class="gain">+${f.gain} XP${f.mult > 1 ? ` · Combo ×${f.mult}` : ''}</span>${q.sub ? `<p class="tr">= ${esc(q.sub)}</p>` : ''}</div>${next}</div>`;
  45    }
  46    if (f.kind === 'almost') {
  47      return `<div class="fb almost" role="status">${mascot('hmm')}<div class="msg"><b>Fast! Nur die Schreibweise:</b><div class="pen">${esc(q.answer)}</div><span class="gain">+${f.gain} XP</span></div>${next}</div>`;
  48    }
  49    const reveal = q.kind === 'match' ? '' : `<div class="pen">${q.reveal || esc(q.answer)}</div>`;
  50    return `<div class="fb bad" role="status">${mascot('sad')}<div class="msg"><b>Kein Problem, so geht es:</b>${reveal}<div class="why">${esc(q.why || 'Schau dir die Formen noch einmal an, dann klappt es.')}</div></div>${next}</div>`;
  51  }

  77      Object.assign(run, { state: 'ask', fb: null, sel: null, typed: '', peek: false, peeked: false });

  81    function gain(amount) {
  82      if (!amount) return;
  83      run.xp += amount;
  84      const r = addXP(ctx.profile, ctx.app, amount);
  85      if (r.levelUp) run.levelUp = r.levelUp;
  86      if (r.goalHit) run.goalHit = true;
  87      ctx.save();

  93      const fb = run.state === 'fb';

  96      if (fb) foot = feedbackHTML(q, run, ctx);

 110    function settle(kind) {
 111      let amount = 0, mult = 1;
 112      const q = cur();
 113      if (q.retry) {
 114        // Zweiter Versuch: zählt nicht für Genauigkeit und Herzen, bringt aber XP fürs Dazulernen.
 115        if (kind === 'ok' || kind === 'almost') { amount = 5; run.relearned++; }
 116        gain(amount);
 117        run.fb = { kind: amount ? 'ok' : 'bad', gain: amount, mult: 1, retry: true };
 118        run.state = 'fb';
 119        render();
 120        return;
 121      }
 122      if (kind === 'ok') {
 123        run.correct++; run.combo++; run.bestCombo = Math.max(run.bestCombo, run.combo);
 124        mult = multiplier(run.combo); amount = 10 * mult;
 125      } else if (kind === 'almost') {
 126        run.almost++; amount = 5;
 127      } else {
 128        run.wrong++; run.combo = 0; run.hearts--;
 129        if (mode !== 'boss' && q.kind !== 'match' && run.hearts > 0) {
 130          run.questions.push({ ...q, retry: true, options: q.options ? shuffle(q.options) : q.options });
 131        }
 132      }
 133      gain(amount);
 134      run.fb = { kind, gain: amount, mult };
 135      run.state = 'fb';
 136      render();
 137    }

 142      if (!type.grade || (type.canCheck && !type.canCheck(q, run))) return;
 143      const kind = type.grade(q, run, mount);
 144      if (kind) settle(kind);
 145    }
