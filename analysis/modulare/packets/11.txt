   5  /** Zahlenstrahl von -8 bis 8: Start bei a, Sprung um b. */
   6  export function lineDiagram(a = 3, b = -5) {
   7    const X = (v) => 260 + v * 28, r = a + b;
   8    let out = line(X(-8) - 10, 110, X(8) + 10, 110, 'ln');
   9    for (let v = -8; v <= 8; v++) {
  10      out += line(X(v), 104, X(v), 116, 'ln-soft');
  11      if (v % 2 === 0) out += text(X(v), 138, fmt(v), `sm mono${v === 0 ? '' : ' mu'}`);
  12    }
  13    out += arrow(X(a), 70, X(r), 70);
  14    out += text((X(a) + X(r)) / 2, 54, `${b > 0 ? '+' : '−'}${Math.abs(b)}`, 'md mono');
  15    out += circle(X(a), 110, 9, 'box') + circle(X(r), 110, 10, 'hi');
  16    out += text(X(a), 175, `Start ${fmt(a)}`, 'sm mu') + text(X(r), 175, `Ziel ${fmt(r)}`, 'md mono');
  17    return svg(520, 210, out, `Zahlenstrahl: Start bei ${fmt(a)}, ${b > 0 ? 'nach rechts' : 'nach links'} um ${Math.abs(b)}, Ziel ${fmt(r)}`);
  18  }

  21  export function tilesDiagram(a1 = 2, c1 = 3, a2 = 1, c2 = 2) {
  22    const tile = (x, y, cls, label) => rect(x, y, 34, 34, cls, 6) + text(x + 17, y + 24, label, `md mono${cls === 'hi' ? ' on-sun' : ''}`);
  23    let out = '';
  24    const group = (x0, y0, a, c) => {
  25      for (let i = 0; i < a; i++) out += tile(x0 + i * 40, y0, 'box', 'x');
  26      for (let i = 0; i < c; i++) out += tile(x0 + i * 40, y0 + 42, 'hi', '1');
  27    };
  28    group(20, 30, a1, c1);
  29    out += text(20 + Math.max(a1, c1) * 40 + 14, 62, '+', 'lg');
  30    group(20 + Math.max(a1, c1) * 40 + 44, 30, a2, c2);
  31    const A = a1 + a2, C = c1 + c2, y = 140;
  32    out += text(260, 128, 'zusammenfassen', 'sm mu') + arrow(260, 132, 260, 146);
  33    const w = Math.max(A, C) * 40;
  34    for (let i = 0; i < A; i++) out += tile(260 - w / 2 + i * 40, y + 20, 'box', 'x');
  35    for (let i = 0; i < C; i++) out += tile(260 - w / 2 + i * 40, y + 62, 'hi', '1');
  36    out += text(260, y + 124, `${a1 === 1 ? '' : a1}x + ${c1} + ${a2 === 1 ? '' : a2}x + ${c2} = ${A}x + ${C}`, 'md mono');
  37    return svg(520, 300, out, `Kacheln: ${a1}x und ${c1} plus ${a2}x und ${c2} ergibt ${A}x und ${C}`);

  47    out += rect(40, 252, 180, 50, 'box') + text(130, 285, `${a}x = ${c - b}`, 'lg mono');
  48    out += rect(300, 252, 180, 50, 'box') + text(390, 285, `| : ${a}`, 'md mono');
  49    out += arrow(222, 277, 298, 277);
  50    out += rect(190, 318, 140, 50, 'hi') + text(260, 351, `x = ${x}`, 'lg mono on-sun');
  51    out += arrow(390, 304, 330, 330);
  52    return svg(520, 384, out, `Waage: ${a}x + ${b} = ${c}. Erst minus ${b}, dann geteilt durch ${a}, Ergebnis x = ${x}`);

  68  /** Wertetabelle mit Pfeilen: proportional (Quotient bleibt gleich) oder antiproportional (Produkt bleibt gleich). */
  69  export function tableDiagram(kind = 'prop', k = 3) {
  70    const xs = [1, 2, 3, 4], total = 12;
  71    const ys = xs.map((x) => (kind === 'prop' ? x * k : total / x));
  72    const prop = kind === 'prop';
  73    let out = text(60, 56, prop ? 'Hefte' : 'Arbeiter', 'sm mu') + text(60, 116, prop ? 'Preis in €' : 'Tage', 'sm mu');
  74    xs.forEach((x, i) => {
  75      const cx = 160 + i * 100;
  76      out += rect(cx - 36, 34, 72, 36, 'box') + text(cx, 59, String(x), 'md mono');
  77      out += rect(cx - 36, 94, 72, 36, i === 1 ? 'hi' : 'box') + text(cx, 119, fmt(ys[i]), `md mono${i === 1 ? ' on-sun' : ''}`);
  78    });
  79    out += text(260, 168, prop ? 'x 2 oben, x 2 unten: Quotient y : x bleibt gleich' : 'x 2 oben, : 2 unten: Produkt x · y bleibt gleich', 'sm mu');
  80    out += text(260, 196, prop ? `Immer ${k} € pro Heft` : `Immer ${total} Arbeitertage`, 'md mono');
  81    return svg(520, 212, out, prop ? `Tabelle proportional: Preis ist immer ${k} mal die Anzahl` : `Tabelle antiproportional: Anzahl mal Tage ergibt immer ${total}`);
  82  }

  94  /** Geordnete Werte als Kacheln. Der Median ist hervorgehoben, darunter Mittelwert und Spannweite (berechnet). */
  95  export function dataDiagram(xs = [3, 9, 4, 4, 7]) {
  96    const s = [...xs].sort((a, b) => a - b), n = s.length, mid = (n - 1) / 2;
  97    const w = 56, x0 = 260 - (n * w) / 2;
  98    let out = text(260, 24, 'der Größe nach geordnet', 'sm mu');
  99    s.forEach((v, i) => {
 100      out += rect(x0 + i * w + 3, 40, w - 6, 48, i === mid ? 'hi' : 'box') + text(x0 + i * w + w / 2, 72, String(v), `md mono${i === mid ? ' on-sun' : ''}`);
 101    });
 102    out += text(x0 + mid * w + w / 2, 112, 'Median', 'sm mu');
 103    const mean = s.reduce((a, b) => a + b, 0) / n;
 104    out += text(260, 150, `Mittelwert = ${s.join(' + ')} : ${n} = ${fmt(mean)}`, 'md mono');
 105    out += text(260, 180, `Spannweite = ${s[n - 1]} \u2212 ${s[0]} = ${s[n - 1] - s[0]}`, 'md mono');
 106    return svg(520, 200, out, `Werte ${s.join(', ')}: Median ${s[mid]}, Mittelwert ${fmt(mean)}, Spannweite ${s[n - 1] - s[0]}`);
 107  }

 122  /** Restschuld Jahr für Jahr: gleichbleibende Tilgung, Zinsen nur auf den Rest. Alle Zahlen werden berechnet. */
 123  export function kreditDiagram(k = 2000, t = 500, p = 4) {
 124    const years = Math.min(Math.round(k / t), 5), W = 70, base = 150, top = 36;
 125    let out = '';
 126    for (let i = 0; i < years; i++) {
 127      const rest = k - i * t, h = Math.round((rest / k) * (base - top)), x = 30 + i * 96;
 128      out += rect(x, base - h, W, h, i === 0 ? 'hi' : 'box', 8);
 129      out += text(x + W / 2, base - h + 24, `${fmt(rest)} €`, `sm mono${i === 0 ? ' on-sun' : ''}`);
 130      out += text(x + W / 2, base + 20, `Jahr ${i + 1}`, 'sm mu');
 131      out += text(x + W / 2, base + 40, 'Zinsen', 'sm mu') + text(x + W / 2, base + 58, `${fmt(rest * p / 100)} €`, 'sm mono');
 132    }
 133    out += text(260, 20, `Restschuld am Jahresanfang, Tilgung ${fmt(t)} € pro Jahr, Zinssatz ${p} %`, 'sm mu');
 134    return svg(520, 218, out, `Kredit ${fmt(k)} Euro, jedes Jahr ${fmt(t)} Euro getilgt: Die Restschuld sinkt, darum sinken auch die Zinsen`);
 135  }
