  20  function addSub() {
  21    const a = nz(-12, 12), b = nz(-12, 12), sub = Math.random() < 0.5;
  22    const r = sub ? a - b : a + b;
  23    const expr = `${fmt(a)} ${sub ? MINUS : '+'} ${par(b)}`;
  24    const wrong = sub ? [a + b, -(a - b), Math.abs(a) - Math.abs(b)] : [-(a + b), Math.abs(a) + Math.abs(b), a - b];
  25    return { prompt: expr, answer: fmt(r), value: r, numeric: true, wrong: nearby(r, wrong).map(fmt),
  26      why: sub ? `Subtrahieren heißt Gegenzahl addieren: ${fmt(a)} + ${par(-b)} = ${fmt(r)}.` : `${fmt(a)} + ${par(b)} = ${fmt(r)}. Bei verschiedenen Vorzeichen rechnest du den Unterschied der Beträge.` };
  27  }

  29  function mulDiv() {
  30    const a = nz(-9, 9), b = nz(-9, 9), div = Math.random() < 0.4;
  31    const r = div ? a : a * b;
  32    const expr = div ? `${fmt(a * b)} : ${par(b)}` : `${par(a)} · ${par(b)}`;
  33    const wrong = [-r, Math.abs(r), a + b];
  34    return { prompt: expr, answer: fmt(r), value: r, numeric: true, wrong: nearby(r, wrong).map(fmt),
  35      why: `${(a < 0) === (b < 0) ? 'Gleiche' : 'Verschiedene'} Vorzeichen ergeben ${(a < 0) === (b < 0) ? 'Plus' : 'Minus'}, also ${fmt(r)}.` };
  36  }

  63  function ausklammern() {
  64    const g = pick([2, 3, 4, 5, 6]), a = rnd(2, 5), c = rnd(2, 6), A = g * a, C = g * c;
  65    const right = `${g} · (${linear(a, c)})`;
  66    return { prompt: `Klammere den größten gemeinsamen Faktor aus: ${linear(A, C)}`, answer: right,
  67      wrong: [`${g} · (${linear(a, C)})`, `${g * 2} · (${linear(a, c)})`, `${a} · (${linear(g, c)})`, `${g} · (${linear(A, c)})`],
  68      choiceOnly: true, label: 'Ausklammern',
  69      why: `Gemeinsamer Faktor von ${A} und ${C} ist ${g}. Teile jeden Summanden durch ${g}: ${linear(a, c)}. Probe durch Ausmultiplizieren: ${g}·${a}x + ${g}·${c} = ${linear(A, C)}.` };

  72  function solve() {
  73    const x = nz(-9, 9), a = rnd(2, 9), b = nz(-12, 12), c = a * x + b;
  74    const prompt = `${linear(a, b)} = ${fmt(c)}`;
  75    const wrong = [(c + b) / a, c - b, x * -1, (c - b) * a];
  76    return { prompt, answer: fmt(x), value: x, numeric: true, wrong: nearby(x, wrong.filter(Number.isInteger)).map(fmt),
  77      why: `${b < 0 ? '+' + Math.abs(b) : MINUS + b}: ${a}x = ${fmt(c - b)}. Dann : ${a}, und es folgt x = ${fmt(x)}. Probe: ${a}·${par(x)} ${b < 0 ? MINUS : '+'} ${Math.abs(b)} = ${fmt(c)}.`, label: 'Löse nach x auf' };
  78  }

  80  function anzahlLoesungen() {
  81    const k = rnd(2, 5), c = rnd(1, 6), d = rnd(1, 6) + c;
  82    const cases = [
  83      [`${linear(k, c)} = ${linear(k, d)}`, 'keine Lösung', `Auf beiden Seiten steht ${k}x. Zieht man es ab, bleibt ${c} = ${d}. Das ist falsch, also gibt es keine Lösung.`],
  84      [`${k} · (x + ${c}) = ${linear(k, k * c)}`, 'unendlich viele Lösungen', `Links ausmultiplizieren ergibt ${linear(k, k * c)}, genau die rechte Seite. Die Gleichung stimmt für jedes x.`],
  85      [`${linear(k, c)} = ${k * (c + 1) + c}`, 'genau eine Lösung', `Hier steht links ein x-Term und rechts nur eine Zahl. Das ergibt genau einen Wert für x.`]
  86    ];
  87    const [eq, ans, why] = pick(cases);
  88    const all = ['keine Lösung', 'genau eine Lösung', 'unendlich viele Lösungen', 'genau zwei Lösungen'];
  89    return { prompt: eq, answer: ans, wrong: all.filter((o) => o !== ans), choiceOnly: true, label: 'Wie viele Lösungen hat die Gleichung?', why };
  90  }

  92  function aufstellen() {
  93    const k = rnd(2, 6), c = rnd(2, 9), x = rnd(2, 8), r = k * x + c;
  94    const eq = `${k}x + ${c} = ${r}`;
  95    return { prompt: `Das ${k}-Fache einer Zahl, vermehrt um ${c}, ergibt ${r}. Welche Gleichung gehört dazu?`, answer: eq,
  96      wrong: [`${k + c}x = ${r}`, `${k}x ${MINUS} ${c} = ${r}`, `${c}x + ${k} = ${r}`, `${k}(x + ${c}) = ${r}`], choiceOnly: true, label: 'Gleichung aufstellen',
  97      why: `„Das ${k}-Fache einer Zahl“ ist ${k}x. „Vermehrt um ${c}“ heißt + ${c}. „Ergibt ${r}“ ist = ${r}. Lösung: x = ${x}.` };
  98  }

 100  function firstStep() {
 101    const a = rnd(2, 9), b = rnd(2, 15), x = rnd(2, 9), c = a * x + b, plus = Math.random() < 0.5;
 102    const eq = plus ? `${a}x + ${b} = ${c}` : `${a}x ${MINUS} ${b} = ${a * x - b}`;
 103    const step = plus ? `| ${MINUS} ${b}` : `| + ${b}`;
 104    return { prompt: eq, answer: step, wrong: [`| : ${a}`, plus ? `| + ${b}` : `| ${MINUS} ${b}`, `| · ${a}`],
 105      why: `Zuerst wird das ${plus ? 'Plus' : 'Minus'} neben dem x rückgängig gemacht (${step}). Das Teilen durch ${a} kommt danach.`, label: 'Welcher Umformungsschritt kommt zuerst?', choiceOnly: true };
 106  }
