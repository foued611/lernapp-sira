   9  function lampBoard(bits, { groups = false } = {}) {
  10    const lamps = WEIGHTS.map((w, i) => `<button type="button" class="bit${bits[i] ? ' on' : ''}${groups && i === 4 ? ' gap' : ''}" data-i="${i}" aria-pressed="${bits[i] ? 'true' : 'false'}" aria-label="Stellenwert ${w}"><span class="w">${w}</span><span class="bulb">${bits[i]}</span></button>`).join('');
  11    return `<div class="bits">${lamps}</div>`;
  12  }

  14  /** Lampen-Tafel, die auf Klicks reagiert. onChange bekommt die Bits und füllt den Ergebnisbereich. */
  15  function mountLamps(root, initial, onChange, options) {
  16    let bits = initial.slice();
  17    const board = root.querySelector('[data-board]');
  18    const paint = () => {
  19      board.innerHTML = lampBoard(bits, options);
  20      onChange(bits);
  21    };
  22    board.addEventListener('click', (e) => {
  23      const lamp = e.target.closest('.bit');
  24      if (!lamp) return;
  25      const i = Number(lamp.dataset.i);
  26      bits[i] = bits[i] ? 0 : 1;
  27      paint();
  28      board.querySelector(`.bit[data-i="${i}"]`).focus();
  29    });
  30    root.querySelectorAll('[data-set]').forEach((btn) => btn.addEventListener('click', () => {
  31      bits = btn.dataset.set === 'all' ? Array(8).fill(1) : Array(8).fill(0);
  32      paint();
  33    }));
  34    paint();
  35  }

 132  /** Standard-Theorie plus ein Spielzeug je Seite. Lektionen und Merkzettel kommen aus content/ (ctx). */
 133  export function renderTheory(box, unit, { lessons, notes: allNotes }) {
 134    const spec = lessons[unit.id];
 135    const notes = (allNotes[unit.id] || []).map((t) => `<div class="note"><b>${esc(t[0])}</b>${esc(t[1])}</div>`).join('');
 136    box.innerHTML = `
 137      <div style="display:grid;gap:22px">
 138        <section class="card tcard"><div class="thead"><h1>${esc(unit.title)}</h1><span class="de">${esc(unit.de)}</span></div></section>
 139        ${lessonTop(spec)}
 140        <div data-toy></div>
 141        ${lessonBottom(spec)}
 142        <section><h2>Merkzettel</h2><div class="notes" style="margin-top:12px">${notes}</div></section>
 143        <div><button class="btn" data-act="start" data-id="${esc(unit.id)}">Übung starten · 8 Aufgaben</button></div>
 144      </div>`;
 145    const slot = box.querySelector('[data-toy]');
 146    if (TOYS[unit.id]) TOYS[unit.id](slot);
 147    wireLesson(box, spec);
 148  }
